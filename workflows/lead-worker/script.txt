// ===== 模型池：DSH llm-pi-ai（sensenova + bai），共 11 个 =====
const DEFAULT_POOL = [
  { provider: "sensenova", model: "deepseek-v4-flash",           label: "商汤 V4 Flash" },
  { provider: "sensenova", model: "glm-5.2",                     label: "GLM 5.2" },
  { provider: "sensenova", model: "sensenova-6.8-flash-lite",    label: "商汤 6.8 Flash Lite" },
  { provider: "sensenova", model: "deepseek-v4-pro",             label: "DeepSeek V4 Pro" },
  { provider: "sensenova", model: "kimi-k3",                     label: "Kimi K3" },
  { provider: "bai",       model: "hy3",                         label: "Hy3" },
  { provider: "bai",       model: "glm-5.3-flash",              label: "GLM 5.3 Flash" },
  { provider: "bai",       model: "qwen3.8-flash",              label: "Qwen3.8 Flash" },
  { provider: "bai",       model: "deepseek-v4-flash",          label: "BAI V4 Flash" },
  { provider: "bai",       model: "mimo-v2.5",                  label: "Mimo V2.5" },
  { provider: "bai",       model: "deepseek-v4-flash-vision-exp", label: "BAI V4 Flash Vision" },
];

// 以 provider+model 作为唯一标识去重，避免组长/组员撞车
const key = (m) => `${m.provider}:${m.model}`;

// 从池中随机抽取 n 个不重复元素
function pickRandom(pool, n) {
  const arr = [...pool];
  const out = [];
  while (out.length < n && arr.length > 0) {
    const i = Math.floor(Math.random() * arr.length);
    out.push(arr.splice(i, 1)[0]);
  }
  return out;
}

// ===== 入参（args 未传时使用默认值 / 占位）=====
const POOL = (args && Array.isArray(args.pool) && args.pool.length) ? args.pool : DEFAULT_POOL;
const WORKER_COUNT = (args && Number.isInteger(args.workerCount)) ? args.workerCount : 3;
const TASK = (args && args.task && String(args.task).trim())
  ? String(args.task).trim()
  : "（占位：请在调用时的 args.task 里填写总任务）";
// 失败子任务的最大重分配轮数（组长每轮重新分配一次）
const MAX_RETRY_ROUNDS = (args && Number.isInteger(args.maxRetryRounds)) ? Math.max(1, args.maxRetryRounds) : 2;

// 组长 1 个（随机），组员 WORKER_COUNT 个（随机，排除组长）
const [组长] = pickRandom(POOL, 1);
const 剩余池 = POOL.filter((m) => key(m) !== key(组长));
const 组员们 = pickRandom(剩余池, Math.min(WORKER_COUNT, 剩余池.length));

// 组员输出有效性：agent() 对普通失败解析为 null（而不是抛错），
// 因此把 null / undefined / 纯空白文本都视为该子任务失败。
const 输出有效 = (r) => r !== null && r !== undefined && !(typeof r === "string" && r.trim() === "");

// ① 组长：拆解总任务为可并行子任务
phase("组长拆解分配");
const 组长拆解 = await agent(
  `你是组长（${组长.label}，模型 ${组长.model}）。\n` +
  `总任务：${TASK}\n` +
  `请把总任务拆解为 ${组员们.length} 个可并行子任务，输出 JSON（键为 tasks）。\n` +
  `tasks 为数组，每项为 { id, title, hint }；hint 是给组员的执行提示。`,
  {
    provider: 组长.provider,
    model: 组长.model,
    label: `组长-拆解分配 [${组长.provider}/${组长.model}]`,
    schema: {
      type: 'object',
      properties: {
        tasks: {
          type: 'array',
          items: {
            type: 'object',
            properties: {
              id: { type: 'string' },
              title: { type: 'string' },
              hint: { type: 'string' },
            },
            required: ['id', 'title', 'hint'],
            additionalProperties: false,
          },
        },
      },
      required: ['tasks'],
      additionalProperties: false,
    },
  },
);

const 子任务 = (组长拆解 && Array.isArray(组长拆解.tasks) && 组长拆解.tasks.length)
  ? 组长拆解.tasks
  : 组员们.map((_, i) => ({ id: `t${i + 1}`, title: `子任务 ${i + 1}`, hint: '按总任务自行处理' }));

// ② 组员并行执行 + 失败重试（失败子任务由组长重新分配，组长只做分配决策，不代写）
// pending 携带每个未完成子任务的状态：尝试次数 / 上次失败的组员下标 / 上次失败原因。
let pending = 子任务.map((t, idx) => ({ idx, t, attempts: 0, lastFailedWorker: -1, lastReason: '尚未执行' }));
const 完成 = new Map(); // 子任务下标 → 组员输出
let round = 0;
while (pending.length > 0 && round <= MAX_RETRY_ROUNDS) {
  // 本轮分配表：子任务下标 → 组员模型
  const 分配 = new Map();
  if (round === 0) {
    phase('组员并行执行');
    // 首轮：按序循环分配给各组员
    pending.forEach((p, k) => 分配.set(p.idx, 组员们[k % 组员们.length]));
  } else {
    phase(`失败重分配（第 ${round} 轮）`);
    // 组长重新分配：把失败子任务指派给组员重试，并给出改进后的执行提示。
    const 重分配 = await agent(
      `你是组长（${组长.label}，模型 ${组长.model}）。\n` +
      `总任务：${TASK}\n\n` +
      `以下子任务在上一轮由组员执行失败。请你重新分配这些子任务给组员重试；` +
      `你只做分配决策，绝不执行子任务本身。\n\n` +
      `可用组员（worker 下标从 0 开始）：\n` +
      组员们.map((m, i) => `${i}: ${m.label}（${m.provider}/${m.model}）`).join('\n') +
      `\n\n失败情况：\n` +
      pending.map((p) => JSON.stringify({
        子任务: p.t,
        已尝试次数: p.attempts,
        上次失败组员下标: p.lastFailedWorker,
        上次失败原因: p.lastReason,
      })).join('\n') +
      `\n\n要求：为每个失败子任务选择一个组员（worker 为其下标），尽量避开其上次失败的组员；` +
      `并在 hint 中针对上次失败原因给出改进后的执行提示。输出 JSON：键 assignments，每项 { id, worker, hint }。`,
      {
        provider: 组长.provider,
        model: 组长.model,
        label: `组长-失败重分配（第 ${round} 轮） [${组长.provider}/${组长.model}]`,
        schema: {
          type: 'object',
          properties: {
            assignments: {
              type: 'array',
              items: {
                type: 'object',
                properties: {
                  id: { type: 'string' },
                  worker: { type: 'integer' },
                  hint: { type: 'string' },
                },
                required: ['id', 'worker'],
                additionalProperties: false,
              },
            },
          },
          required: ['assignments'],
          additionalProperties: false,
        },
      },
    );
    // 应用组长的分配决策；组长调用失败或决策无效时回退为"避开上次失败组员"的轮转分配。
    const byId = new Map(
      (重分配 && Array.isArray(重分配.assignments) ? 重分配.assignments : [])
        .map((a) => [String(a.id), a]),
    );
    pending.forEach((p, k) => {
      const a = byId.get(String(p.t.id));
      let w = (a && Number.isInteger(a.worker)) ? a.worker : -1;
      if (w < 0 || w >= 组员们.length) {
        const candidates = 组员们.map((_, i) => i).filter((i) => i !== p.lastFailedWorker);
        const pool = candidates.length > 0 ? candidates : 组员们.map((_, i) => i);
        w = pool[k % pool.length];
      }
      if (a && typeof a.hint === 'string' && a.hint.trim() !== '') {
        p.t = { ...p.t, hint: a.hint };
      }
      分配.set(p.idx, 组员们[w]);
    });
  }

  // 并行执行本轮分配
  const outcomes = await Promise.all(pending.map(async (p) => {
    const m = 分配.get(p.idx);
    const out = await agent(
      `你是组员（${m.label}）。请完成子任务：${p.t.title}\n执行提示：${p.t.hint}\n只完成该子任务本身，输出该子任务的成果。`,
      { provider: m.provider, model: m.model, label: `组员-${p.t.id} [${m.provider}/${m.model}]` },
    );
    return { p, m, out };
  }));

  // 结算本轮：成功的收下，失败的记录原因进入下一轮重分配
  const nextPending = [];
  for (const { p, m, out } of outcomes) {
    p.attempts += 1;
    if (输出有效(out)) {
      完成.set(p.idx, out);
    } else {
      p.lastFailedWorker = 组员们.indexOf(m);
      p.lastReason = out === null ? '子代理失败（模型返回空）' : '组员返回空内容';
      nextPending.push(p);
    }
  }
  pending = nextPending;
  round += 1;
}

// 轮数耗尽仍未完成的子任务 → 最终失败清单（如实上报，绝不代写）
const 失败子任务 = pending.map((p) => ({
  id: p.t.id,
  title: p.t.title,
  已尝试次数: p.attempts,
  失败原因: p.lastReason,
}));

// ③ 组长：汇总 + 复核（失败子任务如实列出，禁止组长自己补写内容）
phase('组长汇总复核');
const 汇总输入 = 子任务.map((t, i) => JSON.stringify({
  子任务: t,
  状态: 完成.has(i) ? '已完成' : '失败',
  组员输出: 完成.has(i) ? 完成.get(i) : null,
  ...(完成.has(i) ? {} : { 失败原因: (失败子任务.find((f) => f.id === t.id) || {}).失败原因 || '未知' }),
})).join('\n\n');

const 最终交付 = await agent(
  `你是组长（${组长.label}）。以下是各子任务、状态与对应组员输出，请合并、交叉复核，修正明显错误，给出最终交付物。\n\n` +
  `重要约束：状态为"失败"的子任务没有组员输出。你只能在最终交付物中如实列出这些未完成子任务及其失败原因，` +
  `并给出后续建议（suggested_action）；绝不允许你自己代写这些子任务的内容来补齐交付物。\n\n` +
  汇总输入,
  { provider: 组长.provider, model: 组长.model, label: `组长-汇总复核 [${组长.provider}/${组长.model}]` },
);

return {
  组长: `${组长.label}（${组长.model}）`,
  组员: 组员们.map((m) => `${m.label}（${m.model}）`),
  子任务数: 子任务.length,
  完成数: 完成.size,
  失败子任务,
  最终交付,
};
